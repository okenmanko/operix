import type { Response } from 'express';
import { Res } from '@nestjs/common';

@Get('export-excel')
async exportExcel(@CurrentUser() user: AuthUser, @Res() res: Response) {
  const buffer = await this.debtsService.exportExcel(user.companyId);
  res.setHeader('Content-Disposition', 'attachment; filename="operix-debts.xlsx"');
  res.setHeader('Content-Type', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet');
  res.send(buffer);
}

import * as XLSX from 'xlsx';

async exportExcel(companyId: string) {
  const debts = await this.prisma.debt.findMany({
    where: { companyId },
    include: { client: true, payments: true },
    orderBy: { createdAt: 'desc' },
  });

  const rows = debts.map((debt) => ({
    Mijoz: debt.client?.fullName || '',
    Telefon: debt.client?.phone || '',
    Qarz: debt.amount,
    Valyuta: debt.currency,
    Status: debt.status,
    Muddat: debt.dueDate,
    Izoh: debt.comment || '',
  }));

  const sheet = XLSX.utils.json_to_sheet(rows);
  const book = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(book, sheet, 'Debts');
  return XLSX.write(book, { type: 'buffer', bookType: 'xlsx' });
}
